# Glühlabor auf Hetzner Webhosting (Apache). Wird beim Bau als .htaccess in den Hauptordner gelegt.
DirectoryIndex index.html
Options -Indexes
AddType application/manifest+json .webmanifest
AddDefaultCharset UTF-8

<IfModule mod_rewrite.c>
    RewriteEngine On

    # Immer https, immer ohne www
    RewriteCond %{HTTPS} !=on
    RewriteCond %{HTTP:X-Forwarded-Proto} !https [NC]
    RewriteRule ^ https://gluehlabor.de%{REQUEST_URI} [R=301,L]
    RewriteCond %{HTTP_HOST} ^www\. [NC]
    RewriteRule ^ https://gluehlabor.de%{REQUEST_URI} [R=301,L]

    # Manche Webhoster reichen den Authorization-Header nicht an PHP weiter
    RewriteCond %{HTTP:Authorization} .
    RewriteRule .* - [E=HTTP_AUTHORIZATION:%{HTTP:Authorization}]

    # Alles unter /api/ geht an die PHP-Schnittstelle
    RewriteRule ^api/(?!index\.php$).* api/index.php [L,QSA]
</IfModule>

# Dateien, die nie ausgeliefert werden
<FilesMatch "^\.|\.(sqlite|sqlite-wal|sqlite-shm|log|md|sh|txt\.bak)$">
    Require all denied
</FilesMatch>

<IfModule mod_headers.c>
    # Nicht bei Suchmaschinen listen (gilt für alles, auch die Schnittstelle)
    Header always set X-Robots-Tag "noindex, nofollow, noarchive, nosnippet, noimageindex"
    Header always set X-Content-Type-Options "nosniff"
    Header always set Referrer-Policy "no-referrer"
    Header always set X-Frame-Options "DENY"
    Header always set Cross-Origin-Opener-Policy "same-origin"
    Header always set Permissions-Policy "camera=(), microphone=(), geolocation=(), payment=()"
    # Über http wird der Header vom Browser ignoriert, die Umleitung oben schickt ohnehin alles auf https
    Header always set Strict-Transport-Security "max-age=31536000"

    <FilesMatch "\.html$">
        Header always set Content-Security-Policy "default-src 'self'; script-src 'self'; style-src 'self' 'unsafe-inline'; img-src 'self' data: blob:; font-src 'self'; connect-src 'self'; worker-src 'self'; manifest-src 'self'; base-uri 'self'; form-action 'self'; frame-ancestors 'none'; object-src 'none'"
    </FilesMatch>

    # Updates der installierten App dürfen nie an einer alten Kopie hängen bleiben
    <FilesMatch "^(index\.html|sw\.js|registerSW\.js|manifest\.webmanifest)$">
        Header set Cache-Control "no-cache"
    </FilesMatch>
    <If "%{REQUEST_URI} =~ m#^/assets/#">
        Header set Cache-Control "public, max-age=31536000, immutable"
    </If>
</IfModule>

<IfModule mod_deflate.c>
    AddOutputFilterByType DEFLATE text/html text/css text/javascript application/javascript application/json image/svg+xml application/manifest+json
</IfModule>
